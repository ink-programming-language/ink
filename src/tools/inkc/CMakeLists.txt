add_executable(inkc main.cpp module_loader.cpp)
target_compile_features(inkc PRIVATE cxx_std_17)
target_link_libraries(inkc PRIVATE ink::backend ink::cli ink::execution ink::frontend)
target_compile_definitions(inkc PRIVATE "INK_DEFAULT_STD_PATH=\"${PROJECT_SOURCE_DIR}/src/std\"")

if(MSVC)
  target_compile_definitions(inkc PRIVATE "INK_LINKER_PATH=\"${CMAKE_LINKER}\"")
  target_compile_definitions(inkc PRIVATE INK_USE_MSVC_LINKER=1)
  get_filename_component(INK_WINDOWS_SDK_BIN_ARCH_DIRECTORY "${CMAKE_MT}" DIRECTORY)
  get_filename_component(INK_WINDOWS_SDK_VERSION_DIRECTORY "${INK_WINDOWS_SDK_BIN_ARCH_DIRECTORY}" DIRECTORY)
  get_filename_component(INK_WINDOWS_SDK_VERSION "${INK_WINDOWS_SDK_VERSION_DIRECTORY}" NAME)
  get_filename_component(INK_WINDOWS_SDK_BIN_DIRECTORY "${INK_WINDOWS_SDK_VERSION_DIRECTORY}" DIRECTORY)
  get_filename_component(INK_WINDOWS_SDK_ROOT "${INK_WINDOWS_SDK_BIN_DIRECTORY}" DIRECTORY)
  if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(INK_WINDOWS_SDK_ARCHITECTURE x64)
  else()
    set(INK_WINDOWS_SDK_ARCHITECTURE x86)
  endif()
  set(INK_KERNEL32_LIBRARY "${INK_WINDOWS_SDK_ROOT}/Lib/${INK_WINDOWS_SDK_VERSION}/um/${INK_WINDOWS_SDK_ARCHITECTURE}/kernel32.lib")
  if(NOT EXISTS "${INK_KERNEL32_LIBRARY}")
    message(FATAL_ERROR "inkc requires the Windows SDK kernel32 import library at '${INK_KERNEL32_LIBRARY}'")
  endif()
  target_compile_definitions(inkc PRIVATE "INK_KERNEL32_LIBRARY_PATH=\"${INK_KERNEL32_LIBRARY}\"")
else()
  target_compile_definitions(inkc PRIVATE "INK_LINKER_PATH=\"${CMAKE_CXX_COMPILER}\"")
endif()

add_custom_command(TARGET inkc POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory "${PROJECT_SOURCE_DIR}/src/std" "$<TARGET_FILE_DIR:inkc>/std" VERBATIM)

if(MSVC)
  target_compile_options(inkc PRIVATE /utf-8 /W4 /permissive-)
else()
  target_compile_options(inkc PRIVATE -Wall -Wextra -Wpedantic)
endif()

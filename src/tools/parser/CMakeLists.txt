file(GLOB_RECURSE INK_PARSE_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_executable(ink_parse ${INK_PARSE_SOURCES})
target_compile_features(ink_parse PRIVATE cxx_std_23)
target_link_libraries(ink_parse PRIVATE ink::frontend cxxopts::cxxopts spdlog::spdlog)
target_compile_definitions(ink_parse PRIVATE CXXOPTS_NO_EXCEPTIONS CXXOPTS_NO_REGEX)
ink_disable_exceptions(ink_parse)
set_target_properties(ink_parse PROPERTIES OUTPUT_NAME ink-parse)

if(MSVC)
  target_compile_options(ink_parse PRIVATE /utf-8)
endif()

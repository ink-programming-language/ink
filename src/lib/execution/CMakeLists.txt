file(GLOB_RECURSE INK_EXECUTION_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")
add_library(ink_execution STATIC ${INK_EXECUTION_SOURCES})
add_library(ink::execution ALIAS ink_execution)

target_compile_features(ink_execution PUBLIC cxx_std_20)
target_include_directories(ink_execution PUBLIC "$<BUILD_INTERFACE:${PROJECT_SOURCE_DIR}/src/include>")
target_link_libraries(ink_execution PUBLIC ink::ir)
target_link_libraries(ink_execution PRIVATE ink::libffi ink::llvm)
if(CMAKE_DL_LIBS)
  target_link_libraries(ink_execution PRIVATE ${CMAKE_DL_LIBS})
endif()
ink_disable_exceptions(ink_execution)

if(MSVC)
  target_compile_options(ink_execution PRIVATE /utf-8 /W4 /permissive-)
else()
  target_compile_options(ink_execution PRIVATE -Wall -Wextra -Wpedantic)
endif()
